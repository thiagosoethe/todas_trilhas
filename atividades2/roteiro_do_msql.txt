ETAPA 1 - Criar tabela produto

OT3 - VENDAS E TABELA ASSOCIATIVA
==================================================
Cria a tabela vendas e a tabela associativa vendas_dos_produtos, que liga venda com produto.
Uma venda pode ter muitos produtos.
E também um produto pode aparecer em muitas vendas.


ETAPA 1 - Criar tabela vendas e vendas_dos_produtos
Apague o código da tela, cole este código e clique no raio:

USE mercado;

DROP TABLE IF EXISTS vendas_dos_produtos;
DROP TABLE IF EXISTS vendas;

CREATE TABLE vendas (
    idvenda INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    data_venda DATE
);

CREATE TABLE vendas_dos_produtos (
    idvenda INT,
    idproduto INT,
    quantidade INT,
    CONSTRAINT fkvenda FOREIGN KEY (idvenda)
    REFERENCES vendas(idvenda),
    CONSTRAINT fkproduto FOREIGN KEY (idproduto)
    REFERENCES produto(idproduto),
    PRIMARY KEY (idvenda, idproduto)
);

OT6 - TRIGGERS
==================================================
Criação de triggers, que são ações automáticas no banco.
Exemplo: impedir valor inválido ou salvar dados antes de excluir.


ETAPA 1 - Criar tabelas de vendedor

ETAPA 2 - Apagar triggers antigas
Apague o código da tela, cole este código e clique no raio:

USE mercado;

DROP TRIGGER IF EXISTS trg_bloqueia_reducao_salario;
DROP TRIGGER IF EXISTS trg_bloqueia_salario_inicial_baixo;

ETAPA 3 - Criar triggers
Apague o código da tela, cole este código e clique no raio:

USE mercado;

DELIMITER $$

CREATE TRIGGER trg_bloqueia_reducao_salario
BEFORE UPDATE ON vendedor
FOR EACH ROW
BEGIN
    IF NEW.salario < OLD.salario THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Salario novo nao pode ser menor.';
    END IF;
END$$

CREATE TRIGGER trg_bloqueia_salario_inicial_baixo
BEFORE INSERT ON vendedor
FOR EACH ROW
BEGIN
    IF NEW.salario < 1200 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Salario inicial nao pode ser menor que 1200.';
    END IF;
END$$

CREATE TRIGGER trg_vendedor_desligado
BEFORE DELETE ON vendedor
FOR EACH ROW
BEGIN
    INSERT INTO vendedores_desligados (idvendedor, nome, salario, data_desligamento)
    VALUES (OLD.idvendedor, OLD.nome, OLD.salario, NOW());
END$$

DELIMITER ;

OT7 - PROCEDURES
==================================================
Criação de procedures, que são comandos salvos para executar depois com CALL.
Exemplo: atualizar preço de produtos automaticamente.


ETAPA 1 - Apagar procedures antigas
Apague o código da tela, cole este código e clique no raio:

USE mercado;

DROP PROCEDURE IF EXISTS atualizar_preco_produto;
DROP PROCEDURE IF EXISTS aumentar_preco_todos_produtos;
DROP PROCEDURE IF EXISTS aumentar_preco_por_quantidade_vendida;


ETAPA 2 - Criar procedures
Apague o código da tela, cole este código e clique no raio:

USE mercado;

DELIMITER $$

CREATE PROCEDURE atualizar_preco_produto(
    IN p_idproduto INT,
    IN p_novo_preco DECIMAL(10,2)
)
BEGIN
    UPDATE produto
    SET preco = p_novo_preco
    WHERE idproduto = p_idproduto;
END$$

CREATE PROCEDURE aumentar_preco_todos_produtos()
BEGIN
    UPDATE produto
    SET preco = ROUND(preco * 1.10, 2);
END$$

CREATE PROCEDURE aumentar_preco_por_quantidade_vendida(
    IN p_idproduto INT
)
BEGIN
    DECLARE total_vendido INT DEFAULT 0;

    SELECT COALESCE(SUM(quantidade), 0)
    INTO total_vendido
    FROM vendas_dos_produtos
    WHERE idproduto = p_idproduto;

    UPDATE produto
    SET preco = ROUND(preco * (1 + total_vendido / 100), 2)
    WHERE idproduto = p_idproduto;
END$$

DELIMITER ;
